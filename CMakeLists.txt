cmake_minimum_required(VERSION 3.16)
project(security-baseline-scanner VERSION 1.0.0 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "security-baseline-scanner supports Linux only")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(GNUInstallDirs)
set(CMAKE_INSTALL_DEFAULT_DIRECTORY_PERMISSIONS
    OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
add_library(sbs_core src/source.cpp src/rules.cpp src/report.cpp)
target_include_directories(sbs_core PUBLIC include)
target_compile_options(sbs_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(security-baseline-scanner src/main.cpp)
target_link_libraries(security-baseline-scanner PRIVATE sbs_core)
target_compile_options(security-baseline-scanner PRIVATE -Wall -Wextra -Wpedantic -Werror)
include(CTest)
if(BUILD_TESTING)
  add_executable(unit-tests tests/unit.cpp)
  target_link_libraries(unit-tests PRIVATE sbs_core)
  target_compile_options(unit-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME unit COMMAND unit-tests)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:security-baseline-scanner> ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures)
  add_executable(fd-lifecycle-tests tests/fd_lifecycle.cpp)
  target_link_libraries(fd-lifecycle-tests PRIVATE sbs_core)
  target_compile_options(fd-lifecycle-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME fd-lifecycle COMMAND fd-lifecycle-tests)
  set_tests_properties(unit integration fd-lifecycle PROPERTIES TIMEOUT 60)
endif()
install(TARGETS security-baseline-scanner RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES docs/security-baseline-scanner.1 DESTINATION ${CMAKE_INSTALL_MANDIR}/man1)
install(FILES README.md LICENSE docs/CHECKS.md docs/ARCHITECTURE.md docs/report.schema.json DESTINATION ${CMAKE_INSTALL_DOCDIR})
install(FILES debian/copyright DESTINATION ${CMAKE_INSTALL_DOCDIR})
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_CONTACT "GuhsHosting <guhshosting@users.noreply.github.com>")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Conservative read-only Linux security baseline observations")
set(CPACK_DEBIAN_PACKAGE_MAINTAINER "GuhsHosting <guhshosting@users.noreply.github.com>")
set(CPACK_DEBIAN_PACKAGE_SECTION "admin")
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_RPM_PACKAGE_LICENSE "MIT")
set(CPACK_RPM_PACKAGE_GROUP "System Environment/Base")
set(CPACK_PACKAGING_INSTALL_PREFIX "/usr")
set(CPACK_PACKAGE_HOMEPAGE_URL "https://github.com/guhshosting/security-baseline-scanner")
set(CPACK_DEBIAN_PACKAGE_HOMEPAGE "https://github.com/guhshosting/security-baseline-scanner")
set(CPACK_RPM_PACKAGE_URL "https://github.com/guhshosting/security-baseline-scanner")
set(CPACK_RPM_PACKAGE_VENDOR "GuhsHosting")
include(CPack)
